cmake_minimum_required(VERSION 3.21)
project(DazStudioZhCN LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Widgets Quick QuickWidgets Qml)
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
set_target_properties(minhook PROPERTIES AUTOMOC OFF)
add_library(dazzhcn MODULE dazzhcn.cpp plugin.h paintadapter.h graphicstext.h dazzhcn.json)
target_link_libraries(dazzhcn PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::QuickWidgets Qt6::Qml minhook)
target_compile_options(dazzhcn PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8 /Zc:__cplusplus /permissive->)
set_target_properties(dazzhcn PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/generic")
